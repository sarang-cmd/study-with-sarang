name: Deploy landing page and web app to GitHub Pages

on:
  push:
    branches: [main]
    paths: ['landing/**', 'web/**', '.github/workflows/pages.yml']
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
          cache-dependency-path: web/package-lock.json
      - name: Build the web app
        working-directory: web
        run: |
          npm ci
          npm run build
      - name: Assemble the site (landing page at the root, web app under /app/)
        run: |
          mkdir -p _site/app
          cp -r landing/. _site/
          cp -r web/dist/. _site/app/
          touch _site/.nojekyll
      - uses: actions/upload-pages-artifact@v3
        with:
          path: _site

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
